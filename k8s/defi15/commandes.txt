# 1. première application des valeurs du défi 15 (le volume de sauvegarde est créé ; hooks de ConfigMap exécutés)

  -f ../helm/cinetint/values-cours.yaml -f values-defi15.yaml \
  --set images.registre=ghcr.io/eldertek \
  --set ingress.host=django.MONGROUPE.csc8567.luxbulb.org \
  --timeout 10m

# 2. le carnet écrit par les hooks
kubectl get configmap cinetint-deploiement -n u-XXX -o jsonpath='{.data}'; echo
kubectl get configmap cinetint-deploiement -n u-XXX -o yaml

# 3. une mise à jour : les hooks tournent AVANT (sauvegarde) et APRÈS (test de fumée, carnet, sauvegarde)

kubectl get jobs,pods -n u-XXX -w

  -f ../helm/cinetint/values-cours.yaml -f values-defi15.yaml \
  --set images.registre=ghcr.io/eldertek \

# 4. les sauvegardes sur le volume dédié (Pod d'essai qui monte le volume)
kubectl run ls-sauvegardes -n u-XXX --rm -it --restart=Never --image=postgres:17-alpine \
  --labels=app.kubernetes.io/instance=cinetint,app.kubernetes.io/component=sauvegarde \
  --overrides='{"spec":{"containers":[{"name":"ls","image":"postgres:17-alpine","command":["ls","-l","/sauvegardes"],"volumeMounts":[{"name":"s","mountPath":"/sauvegardes"}]}],"volumes":[{"name":"s","persistentVolumeClaim":{"claimName":"cinetint-sauvegardes"}}]}}'
